const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');

let gameOver = false;
let hits = 0;
const maxHits = 3;

const messageEl = document.createElement('div');
messageEl.style.position = 'absolute';
messageEl.style.top = '20px';
messageEl.style.width = '100%';
messageEl.style.textAlign = 'center';
messageEl.style.color = '#fff';
messageEl.style.fontSize = '24px';
document.body.appendChild(messageEl);

// Игрок — сигнал
const signal = {
  x: 50,
  y: 550,
  width: 20,
  height: 20,
  color: ' #0ff',
  speed: 5
};

// Цель — дрон
const drone = {
  x: 700,
  y: 300,
  width: 60,
  height: 40,
  color: 'gray'
};

// Птицы — враги
const birds = [];
const birdCount = 5;

// Обработка клавиш
const keys = {};
window.addEventListener('keydown', (e) => { keys[e.key] = true; });
window.addEventListener('keyup', (e) => { keys[e.key] = false; });

// Создаем птиц
for (let i = 0; i < birdCount; i++) {
  birds.push({
    x: Math.random() * 700 + 50,
    y: Math.random() * 200 + 50,
    width: 40,
    height: 30,
    speedX: (Math.random() - 0.5) * 4,
    speedY: (Math.random() - 0.5) * 4,
    color: 'brown'
  });
}

// Создаем изображение для птицы
const birdImage = new Image();
birdImage.src = 'image/птица-converted.jpeg';

// После загрузки изображения запускаем игру
birdImage.onload = () => {
  // Основной цикл
  function gameLoop() {
    if (gameOver) return;
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    moveBirds();
    checkCollisions();
    moveSignal();
    checkArrival();

    // Рисуем птиц изображением
    birds.forEach(b => {
      ctx.drawImage(birdImage, b.x, b.y, b.width, b.height);
    });

    // Рисуем дрон
    ctx.fillStyle = drone.color;
    ctx.fillRect(drone.x, drone.y, drone.width, drone.height);

    // Рисуем сигнал
    ctx.fillStyle = signal.color;
    ctx.fillRect(signal.x, signal.y, signal.width, signal.height);

    // Таймер и сообщение
    ctx.fillStyle = 'white';
    ctx.font = '20px Arial';
    ctx.fillText('Поражений: ' + hits + ' / ' + maxHits, 10, 20);
    ctx.fillText('Долететь до дрона', 10, 50);

    requestAnimationFrame(gameLoop);
  }

  gameLoop();
};

// Обновление птиц
function moveBirds() {
  birds.forEach(bird => {
    bird.x += bird.speedX;
    bird.y += bird.speedY;
    if (bird.x < 0 || bird.x + bird.width > canvas.width) bird.speedX *= -1;
    if (bird.y < 0 || bird.y + bird.height > canvas.height) bird.speedY *= -1;
  });
}

// Проверка столкновений сигнала с птицами
function checkCollisions() {
  for (let bird of birds) {
    if (
      signal.x < bird.x + bird.width &&
      signal.x + signal.width > bird.x &&
      signal.y < bird.y + bird.height &&
      signal.y + signal.height > bird.y
    ) {
      hits++;
      if (hits >= maxHits) {
        gameOver = true;
        messageEl.innerText = 'Птицы поймали сигнал! Игра окончена.';
      }
    }
  }
}

// Обработка перемещения сигнала
function moveSignal() {
  if (keys['ArrowUp']) signal.y -= signal.speed;
  if (keys['ArrowDown']) signal.y += signal.speed;
  if (keys['ArrowLeft']) signal.x -= signal.speed;
  if (keys['ArrowRight']) signal.x += signal.speed;
  // границы
  signal.x = Math.max(0, Math.min(canvas.width - signal.width, signal.x));
  signal.y = Math.max(0, Math.min(canvas.height - signal.height, signal.y));
}

// Проверка достижения дрона
function checkArrival() {
  if (
    signal.x < drone.x + drone.width &&
    signal.x + signal.width > drone.x &&
    signal.y < drone.y + drone.height &&
    signal.y + signal.height > drone.y
  ) {
    if (!gameOver && hits < maxHits) {
      gameOver = true;
      messageEl.innerText = 'Вы успешно доставили лекарства! Победа!';
    }
  }
}

<script>
const board = document.getElementById('board');
const timerEl = document.getElementById('timerEl');
const movesEl = document.getElementById('movesEl');
const pairsEl = document.getElementById('pairsEl');
const msgEl = document.getElementById('msgEl');
const resetBtn = document.getElementById('reset');
const winScreen = document.getElementById('winScreen');
const loseScreen = document.getElementById('loseScreen');
const startScreen = document.getElementById('startScreen');
const startBtn = document.getElementById('startBtn');
const backBtn = document.getElementById('backBtn');
const retryBtn = document.getElementById('retryBtn');

const SYMBOLS = ['\u2B50','\u2728','\u2604','\u2665','\u2660','\u2663','\u2666','\u2744','\u2600','\u2601','\u2602','\u2603'];

let cards = [], firstCard = null, secondCard = null, lock = false;
let moves = 0, pairs = 0, timeLeft = 60, timerInterval, gameActive = false;

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
}

function buildBoard() {
  board.innerHTML = '';
  cards = [];
  const deck = [...SYMBOLS, ...SYMBOLS];
  shuffle(deck);
  deck.forEach(sym => {
    const card = document.createElement('div');
    card.className = 'card';
    card.dataset.symbol = sym;
    const inner = document.createElement('div');
    inner.className = 'card-inner';
    const front = document.createElement('div');
    front.className = 'card-face card-front';
    const back = document.createElement('div');
    back.className = 'card-face card-back';
    back.textContent = sym;
    inner.appendChild(front);
    inner.appendChild(back);
    card.appendChild(inner);
    board.appendChild(card);
    cards.push(card);
    card.addEventListener('click', () => onCardClick(card));
  });
}

function onCardClick(card) {
  if (!gameActive || lock) return;
  if (card.classList.contains('flipped') || card.classList.contains('matched')) return;
  if (firstCard && secondCard) return;
  card.classList.add('flipped');
  if (!firstCard) { firstCard = card; return; }
  secondCard = card;
  moves++;
  movesEl.textContent = 'Ходов: ' + moves;
  lock = true;
  if (firstCard.dataset.symbol === secondCard.dataset.symbol) {
    setTimeout(() => {
      firstCard.classList.add('matched');
      secondCard.classList.add('matched');
      firstCard = null; secondCard = null; lock = false;
      pairs++;
      pairsEl.textContent = 'Пары: ' + pairs + '/12';
      msgEl.textContent = '';
      if (pairs === 12) winGame();
    }, 400);
  } else {
    setTimeout(() => {
      firstCard.classList.remove('flipped');
      secondCard.classList.remove('flipped');
      firstCard = null; secondCard = null; lock = false;
    }, 800);
    msgEl.textContent = 'Не совпало!'; msgEl.style.color = '#d61a9b';
    setTimeout(() => { msgEl.textContent = ''; }, 800);
  }
}

function startTimer() {
  clearInterval(timerInterval);
  timerInterval = setInterval(() => {
    if (!gameActive) return;
    timeLeft--;
    timerEl.textContent = 'Время: ' + timeLeft;
    if (timeLeft <= 0) { gameActive = false; clearInterval(timerInterval); loseScreen.classList.add('show'); }
  }, 1000);
}

function startGame() {
  moves = 0; pairs = 0; timeLeft = 60; gameActive = true;
  movesEl.textContent = 'Ходов: 0';
  pairsEl.textContent = 'Пары: 0/12';
  timerEl.textContent = 'Время: 60';
  msgEl.textContent = '';
  winScreen.classList.remove('show');
  loseScreen.classList.remove('show');
  buildBoard();
  startTimer();
}

function winGame() {
  gameActive = false;
  clearInterval(timerInterval);
  const score = Math.max(0, timeLeft * 10 - moves * 3);
  msgEl.innerHTML = 'Победа! Очки: ' + score;
  msgEl.style.color = '#0f8';
  setTimeout(() => { winScreen.classList.add('show'); }, 500);
}

startBtn.addEventListener('click', () => { startScreen.classList.remove('show'); startGame(); });
resetBtn.addEventListener('click', startGame);
backBtn.addEventListener('click', () => { location.href = 'index.html'; });
retryBtn.addEventListener('click', () => { loseScreen.classList.remove('show'); startGame(); });

  height: 40,
  color: 'gray'

const birds = [];

// Обработка клавиш

window.addEventListener('keyup', (e) => { keys[e.key] = false; });

  birds.push({
    x: Math.random() * 700 + 50,
    y: Math.random() * 200 + 50,
    width: 40,
    height: 30,
    speedX: (Math.random() - 0.5) * 4,
    speedY: (Math.random() - 0.5) * 4,
    color: 'brown'
  });

birdImage.src = 'image/птица-converted.jpeg';

// После загрузки изображения запускаем игру

  // Основной цикл
  function gameLoop() {
    if (gameOver) return;

    moveBirds();
    checkCollisions();
    moveSignal();
    checkArrival();

    // Рисуем птиц изображением

      ctx.drawImage(birdImage, b.x, b.y, b.width, b.height);
    });

    // Рисуем дрон

    // Рисуем сигнал

    // Таймер и сообщение
    ctx.fillStyle = 'white';
    ctx.font = '20px Arial';
    ctx.fillText('Поражений: ' + hits + ' / ' + maxHits, 10, 20);
    ctx.fillText('Долететь до дрона', 10, 50);

    requestAnimationFrame(gameLoop);
  }

  gameLoop();
};

// Обновление птиц
function moveBirds() {
  birds.forEach(bird => {
    bird.x += bird.speedX;
    bird.y += bird.speedY;
    if (bird.x < 0 || bird.x + bird.width > canvas.width) bird.speedX *= -1;
    if (bird.y < 0 || bird.y + bird.height > canvas.height) bird.speedY *= -1;
  });
}

// Проверка столкновений сигнала с птицами
function checkCollisions() {
  for (let bird of birds) {
    if (
      signal.x < bird.x + bird.width &&
      signal.x + signal.width > bird.x &&
      signal.y < bird.y + bird.height &&
      signal.y + signal.height > bird.y
    ) {
      hits++;
      if (hits >= maxHits) {

  signal.y = Math.max(0, Math.min(canvas.height - signal.height, signal.y));
}

      gameOver = true;
      messageEl.innerText = 'Вы успешно доставили лекарства! Победа!';
    }
  }

  <script>
    const video = document.getElementById('introVideo');
    const overlay = document.getElementById('blackOverlay');
    const gameContainer = document.getElementById('gameContainer');
    const videoWrapper = document.getElementById('videoWrapper');
    const soundBtn = document.getElementById('soundBtn');
    const nextLevelBtn = document.getElementById('nextLevelBtn');

    let soundEnabled = false;

    // Кнопка «Включить звук»
    soundBtn.addEventListener('click', () => {
      if (soundEnabled) return;
      soundEnabled = true;
      soundBtn.style.display = 'none';

      video.muted = false;
      video.play().catch(e => {
        console.log('Автовоспроизведение со звуком заблокировано браузером:', e);
      });
    });

    // Когда видео закончится — показываем оверлей
    video.onended = () => {
      overlay.classList.add('show');
      overlay.style.display = 'flex';
    };

    // Кнопка «Перейти дальше»: сначала показываем игру, потом переходим
    nextLevelBtn.addEventListener('click', () => {
      // Скрываем видео и оверлей
      videoWrapper.style.display = 'none';
      overlay.style.display = 'none';

      // Переход на следующую страницу через 0.5 секунды
      setTimeout(() => {
        window.location.href = 'level1.html';
      }, 100);
    });
